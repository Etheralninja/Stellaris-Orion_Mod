namespace = "orion"
country_event = {
	id = orion.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		NOT = {
			has_country_flag = orion_civic_flag
		}
		has_civic = orion_civic
	}
	immediate = {
		set_country_flag = orion_civic_flag
		create_leader = {
			name = "Orion Fleet Admiral"
			species = this
			traits = {
				trait = trait_admiral
			}
			class = admiral
		}
		create_fleet = {
			effect = {
				set_fleet_flag = orion_fleet_flag
				set_owner = PREV
				create_ship = {
					random_existing_design = military
					graphical_culture = owner
				}
				set_location = {
					target = owner.capital_scope.solar_system.starbase
				}
				assign_leader = last_created_leader
			}
			name = "Orion Fleet"
		}
	}
}
